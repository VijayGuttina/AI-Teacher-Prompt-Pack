% Commercial V1 PDF layout controls.
% This header is deliberately independent of the DOCX reference template.
% The PDF is generated directly from the canonical Markdown source.

\usepackage{geometry}
% Conservative A4 margins to keep text safely inside the printable area.
\geometry{a4paper,top=20mm,bottom=20mm,left=22mm,right=22mm,headheight=10mm,headsep=6mm,footskip=12mm}

\usepackage{microtype}
\microtypesetup{protrusion=false,expansion=false}

\usepackage{xurl}
\usepackage{fvextra}
\usepackage{needspace}

% XeLaTeX/LuaLaTeX Unicode-safe font handling.
% Do not use fontenc here. With XeLaTeX it can cause Unicode characters
% such as curly apostrophes to fall back to EC fonts and produce warnings.
\usepackage{fontspec}
\setmainfont{Latin Modern Roman}
\setsansfont{Latin Modern Sans}
\setmonofont{Latin Modern Mono}

% Latin Modern does not provide every Unicode symbol used in the source.
% Map the small set of symbols used by the book to stable TeX equivalents.
\usepackage{newunicodechar}
\newunicodechar{↪}{\ensuremath{\hookrightarrow}}
\newunicodechar{→}{\ensuremath{\rightarrow}}
\newunicodechar{⌋}{\ensuremath{\rfloor}}

% Give ordinary paragraphs enough flexibility to wrap cleanly inside the
% text area. This is important for normal prose as well as prompt blocks.
\setlength{\emergencystretch}{4em}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.35em}
\tolerance=9999
\pretolerance=1000
\sloppy
\raggedbottom

% Long copy-and-paste blocks wrap at normal word boundaries. breakanywhere=true
% was intentionally removed because fvextra displays a hook arrow at each
% forced break, which was appearing as "↪" inside every prompt in the PDF.
\RecustomVerbatimEnvironment{Highlighting}{Verbatim}{
  breaklines=true,
  breakanywhere=false,
  breaksymbolleft={},
  breaksymbolright={},
  fontsize=\small,
  commandchars=\\\{\}
}

\RecustomVerbatimEnvironment{Verbatim}{Verbatim}{
  breaklines=true,
  breakanywhere=false,
  breaksymbolleft={},
  breaksymbolright={},
  fontsize=\small
}
